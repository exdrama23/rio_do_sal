"use client"

import { useRef, useMemo } from "react"
import { useFrame, useThree } from "@react-three/fiber"
import * as THREE from "three"

const vertexShader = `
varying vec2 vUv;
void main() {
  vUv = uv;
  gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
}
`

const fragmentShader = `
uniform float uTime;
uniform vec2 uResolution;
uniform vec2 uMouse;
uniform vec2 uScroll;

varying vec2 vUv;

float random(vec2 st) {
  return fract(sin(dot(st.xy, vec2(12.9898, 78.233))) * 43758.5453123);
}

float noise(vec2 p) {
  vec2 i = floor(p);
  vec2 f = fract(p);
  f = f * f * (3.0 - 2.0 * f);
  float a = random(i);
  float b = random(i + vec2(1.0, 0.0));
  float c = random(i + vec2(0.0, 1.0));
  float d = random(i + vec2(1.0, 1.0));
  return mix(mix(a, b, f.x), mix(c, d, f.x), f.y);
}

float fbm(vec2 p) {
  float value = 0.0;
  float amp = 0.5;
  float freq = 1.0;
  for (int i = 0; i < 5; i++) {
    value += amp * noise(p * freq);
    freq *= 2.0;
    amp *= 0.5;
  }
  return value;
}

void main() {
  vec2 uv = vUv;
  float t = uTime * 0.12;

  float wave1 = sin(uv.x * 8.0 + t) * 0.03;
  float wave2 = sin(uv.y * 6.0 - t * 0.7) * 0.03;

  vec2 dUv = uv + vec2(wave1, wave2);

  float n = fbm(dUv * 3.0 + t * 0.2);
  n += uScroll.y * 0.1;

  float mi = 0.0;
  if (uMouse.x > 0.0) {
    vec2 m = uMouse / uResolution;
    float d = distance(dUv, m);
    mi = 0.15 * exp(-d * 6.0);
  }
  n += mi;

  vec3 c1 = vec3(0.85, 0.55, 0.04);
  vec3 c2 = vec3(0.65, 0.30, 0.02);
  vec3 c3 = vec3(0.10, 0.07, 0.04);

  vec3 color = mix(c1, c2, n);
  color = mix(color, c3, smoothstep(0.4, 0.8, n));

  float a = smoothstep(0.05, 0.35, n);

  gl_FragColor = vec4(color, a * 0.3);
}
`

export default function ShaderPlane() {
  const ref = useRef<THREE.Mesh>(null!)
  const mouse = useRef(new THREE.Vector2(0, 0))
  const scroll = useRef(new THREE.Vector2(0, 0))
  const { size, gl } = useThree()

  useMemo(() => {
    const onPointerMove = (e: PointerEvent) => {
      mouse.current.x = e.clientX
      mouse.current.y = window.innerHeight - e.clientY
    }
    const onScroll = () => {
      scroll.current.y = window.scrollY / (document.body.scrollHeight - window.innerHeight)
    }
    gl.domElement.addEventListener("pointermove", onPointerMove)
    window.addEventListener("scroll", onScroll, { passive: true })
    return () => {
      gl.domElement.removeEventListener("pointermove", onPointerMove)
      window.removeEventListener("scroll", onScroll)
    }
  }, [gl])

  const uniforms = useMemo(
    () => ({
      uTime: { value: 0 },
      uResolution: { value: new THREE.Vector2(size.width, size.height) },
      uMouse: { value: new THREE.Vector2(0, 0) },
      uScroll: { value: new THREE.Vector2(0, 0) },
    }),
    [size.width, size.height],
  )

  useFrame(({ clock }) => {
    uniforms.uTime.value = clock.getElapsedTime()
    uniforms.uMouse.value.lerp(mouse.current, 0.05)
    uniforms.uScroll.value.lerp(scroll.current, 0.05)
  })

  return (
    <mesh ref={ref} scale={[8, 6, 1]}>
      <planeGeometry args={[1, 1]} />
      <shaderMaterial
        vertexShader={vertexShader}
        fragmentShader={fragmentShader}
        uniforms={uniforms}
        transparent
        depthWrite={false}
      />
    </mesh>
  )
}
